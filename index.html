name: Mirror and deploy adaptambiental on: workflow_dispatch: push: branches: - main jobs: fetch-and-deploy: runs-on: ubuntu-latest steps: - name: Checkout repo uses: actions/checkout@v4 - name: Install wget run: sudo apt-get update && sudo apt-get install -y wget - name: Mirror site with wget run: | rm -rf site mkdir -p site # Mirror the /lander page and its assets into site/www.adaptambiental.com.br wget --mirror --convert-links --adjust-extension --page-requisites --no-parent https://www.adaptambiental.com.br/lander -P site - name: Normalize output directory (move lander content to publish root) run: | # wget will create site/www.adaptambiental.com.br/lander ROOT_DIR="site/www.adaptambiental.com.br" LANDER_DIR="$ROOT_DIR/lander" if [ -d "$LANDER_DIR" ]; then echo "Found lander dir; moving its contents to publish root" mkdir -p "$ROOT_DIR" # move files (if any), preserve hidden files too shopt -s dotglob || true mv -f "$LANDER_DIR"/* "$ROOT_DIR/" || true rm -rf "$LANDER_DIR" fi # fallback in case wget structure differs if [ -d "$ROOT_DIR" ]; then PUBLISH_DIR="$ROOT_DIR" else PUBLISH_DIR=$(find site -maxdepth 2 -type d -name 'www.adaptambiental.com.br*' -print | head -n 1) PUBLISH_DIR=${PUBLISH_DIR:-site} fi echo "PUBLISH_DIR=${PUBLISH_DIR}" ls -la "$PUBLISH_DIR" || true echo "www.adaptambiental.com.br" > "$PUBLISH_DIR/CNAME" echo "$PUBLISH_DIR" > publish_dir.txt - name: Deploy to gh-pages uses: peaceiris/actions-gh-pages@v3 with: github_token: ${{ secrets.GITHUB_TOKEN }} publish_dir: ./site/www.adaptambiental.com.br publish_branch: gh-pages
